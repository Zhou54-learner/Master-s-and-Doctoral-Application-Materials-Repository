def Dict_open():#将dict.txt中存储的数据传入字典中
    dic={}
    with open('dict.txt','r') as f:
        for line in f.readlines():
            dic[tuple(eval(line.split('  ')[0]))]=float(line.split('  ')[1].strip('\n'))
    return dic
def Dict_save(dic,path):#将字典dic存储到path中
    with open(path,'a') as f:
        for key,value in dic.items():
            f.write(str(key)+'  '+str(value)+'\n')

def Initialization(pop_size,dna_size):#初始化，输入种群大小和DNA大小，返回一个种群列表
    pop=[]
    while len(pop)<pop_size:
        individual=[random.randint(0,1)for _ in range (dna_size)]
        if individual not in pop:
            pop.append(individual)
    print("初始化完成")
    return pop
def Function(pop):
    global dic
    pop_function=[]
    for individual in pop:
        if tuple(individual) in dic.keys():
            pop_function.append(dic[tuple(individual)])
        else:
            pop_function.append(Runcst(individual))
    return pop_function

def Mutation(individual):
    if random.random()<p_m:
        mutate_point1 = np.random.randint(0, dna_size)
        mutate_point2 = np.random.randint(0, dna_size)
        individual[mutate_point1] = 1 - individual[mutate_point1]
        individual[mutate_point2] = 1 - individual[mutate_point2]
    return individual


def Cross(pop,pop_fitness):
    select1 = random.random()
    p = 0
    for i in range(pop_size):
        p += pop_fitness[i] / sum_function
        if select1 < p:
            father = pop[i].copy()
            break
    select2 = random.random()
    p = 0
    for i in range(pop_size):
        p += pop_fitness[i] / sum_function
        if select2 < p:
            mather = pop[i].copy()
            break
    cross_point1 = np.random.randint(0, 9)
    cross_point2 = np.random.randint(9, 18)
    return father[0:cross_point1]+mather[cross_point1:cross_point2]+father[cross_point2:dna_size]

def Newpop(pop):
    global sum_function
    newpop = []
    pop_function = Function(pop)
    with open('linshi.txt', 'a') as f:
        for i in range(len(pop)):
            f.write(str(pop[i]) + '  ' + str(pop_function[i]) + '\n')
    sum_function = sum(pop_function)
    while len(newpop) < pop_size-1:
        newindividual = Mutation(Cross(pop, pop_function)).copy()
        newpop.append(newindividual)
    index = pop_function.index(max(pop_function))#精英策略
    newpop.append(pop[index])
    return newpop
