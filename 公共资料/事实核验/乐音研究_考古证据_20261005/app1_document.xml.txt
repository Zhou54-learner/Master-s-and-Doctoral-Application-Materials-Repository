function ButtonPushed(app, ~)
    % 读取音频文件路径
    audioFilePath = app.EditField.Value;

    % 读取音频文件
    [audio, fs] = audioread(audioFilePath);

    % 设置音高检测参数
    frameSize = 1024;
    hopSize = 512;
    minFreq = 80;
    maxFreq = 10000;

    % 计算音高
    [pitchValues, timeInstants] = pitch(audio, fs, ...
        'Method', 'NCF', ...
        'Range', [minFreq, maxFreq]);

    % 在 App 的坐标轴中绘制音高随时间的变化
    plot(app.UIAxes, timeInstants, pitchValues, '.-');
    xlabel(app.UIAxes, '时间 (s)');
    ylabel(app.UIAxes, '音高 (Hz)');
    title(app.UIAxes, '音高随时间的变化');

    end
        end
    end
