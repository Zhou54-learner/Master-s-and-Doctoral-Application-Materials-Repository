function [xx,yy,vxx,vyy] = xlg(x,y,vx,vy,dt,x01,x02,x03,x04,x05,x06,y01,y02,y03,y04,y05,y06,R0,R1,R2,R3,R4,R5,R6,M0,k)
%下一个时刻的速度和位置
x1=zeros(2,2);
x1(1,1)=x;
x1(1,2)=y;
x1(2,1)=vx;
x1(2,2)=vy;
g1=zeros(2,2);
g1(1,1)=vx;
g1(1,2)=vy;
g1(2,1)=vxnn(x,x01,x02,x03,x04,x05,x06,y,y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
g1(2,2)=vynn(x,x01,x02,x03,x04,x05,x06,y,y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
k1=dt.*g1;
x2=x1+0.5*k1;
g2=zeros(2,2);
g2(1,1)=x2(2,1);
g2(1,2)=x2(2,2);
g2(2,1)=vxnn(x2(1,1),x01,x02,x03,x04,x05,x06,x2(1,2),y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
g2(2,2)=vynn(x2(1,1),x01,x02,x03,x04,x05,x06,x2(1,2),y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
k2=dt.*g2;
x3=x1+0.5*k2;
g3=zeros(2,2);
g3(1,1)=x3(2,1);
g3(1,2)=x3(2,2);
g3(2,1)=vxnn(x3(1,1),x01,x02,x03,x04,x05,x06,x3(1,2),y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
g3(2,2)=vynn(x3(1,1),x01,x02,x03,x04,x05,x06,x3(1,2),y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
k3=dt.*g3;
x4=x1+k3;
g4=zeros(2,2);
g4(1,1)=x4(2,1);
g4(1,2)=x4(2,2);
g4(2,1)=vxnn(x4(1,1),x01,x02,x03,x04,x05,x06,x4(1,2),y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
g4(2,2)=vynn(x4(1,1),x01,x02,x03,x04,x05,x06,x4(1,2),y01,y02,y03,y04,y05,y06,k,R0,R1,R2,R3,R4,R5,R6,M0);
k4=dt.*g4;
x1=x1+(1/6)*(k1+2*k2+2*k3+k4);
xx=x1(1,1);
yy=x1(1,2);
vxx=x1(2,1);
vyy=x1(2,2);
end
