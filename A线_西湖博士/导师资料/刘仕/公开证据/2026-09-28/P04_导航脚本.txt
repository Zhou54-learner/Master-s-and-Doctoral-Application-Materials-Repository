function attrs(attributes) {
return Object.keys(attributes || {}).map(function(key) {
var value = attributes[key];
if (value === true) {
return " " + key;
}
if (value === false || value === null || value === undefined) {
return "";
}
return " " + key + "=\"" + String(value).replace(/"/g, """) + "\"";
}).join("");
}

return "";

options = options || {};

"    " + scriptTag("js/bootstrap.min.js"),

"    " + scriptTag("js/isotope.pkgd.min.js"),
"    " + scriptTag("js/validator.min.js")
];
(options.beforeScripts || []).forEach(function(src) {
lines.push("    " + scriptTag(src));
});
lines.push("    " + scriptTag("js/scripts.js"));
(options.afterScripts || []).forEach(function(src) {

});
